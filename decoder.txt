FONCTION build_function_name_tokens(model, function_definitions: list[FunctionDefinition]) -> dict[str, list[int]]:
    """Construit le mapping nom_de_fonction -> liste de token_ids, avec les VRAIS tokens du modèle."""
    résultat = dictionnaire vide
    POUR chaque fonction DANS function_definitions:
        tensor_tokens = model.encode(fonction.name)
        # le tensor est en 2D, il faut l'aplatir en simple liste d'entiers
        tokens = convertir tensor_tokens en list[int]
        résultat[fonction.name] = tokens
    RETOURNER résultat


FONCTION get_next_allowed_tokens(function_name_tokens: dict[str, list[int]], prefix: list[int]) -> set[int]:
    """Retourne l'ensemble des tokens valides pour continuer, selon ce qui est déjà généré."""
    allowed = ensemble vide
    POUR chaque (nom, tokens) DANS function_name_tokens.items():
        SI tokens[:len(prefix)] == prefix ET len(prefix) < len(tokens):
            allowed.ajouter(tokens[len(prefix)])
    RETOURNER allowed


FONCTION is_function_name_complete(function_name_tokens: dict[str, list[int]], prefix: list[int]) -> str | None:
    """Retourne le nom de fonction si prefix correspond EXACTEMENT à ses tokens complets."""
    POUR chaque (nom, tokens) DANS function_name_tokens.items():
        SI tokens == prefix:
            RETOURNER nom
    RETOURNER None

FONCTION generate_function_name(model, prompt: str, function_name_tokens: dict[str, list[int]]) -> str:
    """Génère token par token le nom de fonction choisi par le modèle,
    en respectant les contraintes de function_name_tokens."""

    # Étape A.1 : encoder le prompt de départ
    prompt_tokens = model.encode(prompt)   # tensor 2D
    prompt_tokens = convertir en liste Python plate  # via [0].tolist()

    generated = liste vide  # les tokens du nom en cours de génération

    # Étape A.2 : boucle de génération contrainte
    TANT QUE is_function_name_complete(function_name_tokens, generated) est None:

        # construire la séquence complète à donner au modèle : prompt + ce qui est généré jusqu'ici
        current_input = prompt_tokens + generated

        # obtenir les logits pour le prochain token
        logits = model.get_logits_from_input_ids(current_input)

        # calculer les tokens autorisés à cet instant
        allowed = get_next_allowed_tokens(function_name_tokens, generated)

        SI allowed est vide:
            # cas anormal (ne devrait pas arriver si function_name_tokens est bien construit)
            LEVER une erreur claire

        # parmi les tokens autorisés, choisir celui avec le plus haut logit
        meilleur_token = argmax sur { token_id : logits[token_id] POUR token_id DANS allowed }

        # ajouter ce token à la séquence en cours de génération
        generated.ajouter(meilleur_token)

    # Étape A.3 : la boucle s'est arrêtée, generated correspond à un nom complet
    nom_choisi = is_function_name_complete(function_name_tokens, generated)
    RETOURNER nom_choisi
